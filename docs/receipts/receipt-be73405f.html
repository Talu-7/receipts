def set_seed(seed: int):
    import random
    import numpy as np
    import torch
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False

def apply_bug_fixes():
    # Fix 1: data_provider/data_factory.py -> set drop_last=False for test
    factory_path = os.path.join(os.getcwd(), "data_provider", "data_factory.py")
    if os.path.isfile(factory_path):
        with open(factory_path, "r") as f:
            lines = f.readlines()
        new_lines = []
        in_test_block = False
        for line in lines:
            stripped = line.strip()
            # Detect start of test flag block
            if stripped.startswith("if flag == 'test':") or stripped.startswith('if flag == "test":'):
                in_test_block = True
                new_lines.append(line)
                continue
            if in_test_block and stripped.startswith("drop_last"):
                # Replace drop_last = True with False, keep indentation
                indent = len(line) - len(line.lstrip())
                new_lines.append(" " * indent + "drop_last = False\n")
                in_test_block = False  # assume only one line after flag
                continue
            # If we encounter another assignment that ends the block, reset flag
            if in_test_block and (stripped.startswith("shuffle_flag") or stripped.startswith("batch_size") or stripped.startswith("freq")):
                # still in block, keep line as is
                new_lines.append(line)
                continue
            if in_test_block and not (stripped.startswith("shuffle_flag") or stripped.startswith("batch_size") or stripped.startswith("freq") or stripped.startswith("drop_last")):
                # likely end of block
                in_test_block = False
            new_lines.append(line)
        # Write back if changed
        if new_lines != lines:
            with open(factory_path, "w") as f:
                f.writelines(new_lines)
            print("Patched data_factory.py: set drop_last=False for test", file=sys.stdout)
    else:
        print("Warning: data_factory.py not found", file=sys.stderr)

    # Fix 2: exp/exp_main_F.py -> replace np.array with np.concatenate for preds, trues, inputx
    exp_path = os.path.join(os.getcwd(), "exp", "exp_main_F.py")
    if os.path.isfile(exp_path):
        with open(exp_path, "r") as f:
            lines = f.readlines()
        new_lines = []
        for line in lines:
            stripped = line.strip()
            if stripped.startswith("preds = np.array(preds)"):
                indent = len(line) - len(line.lstrip())
                new_lines.append(" " * indent + "preds = np.concatenate(preds, axis=0)\n")

                indent = len(line) - len(line.lstrip())

                indent = len(line) - len(line.lstrip())
                new_lines.append(" " * indent + "inputx = np.concatenate(inputx, axis=0)\n")

        if new_lines != lines:

def main():
